Module 02 • Linear Pointer Structures

Singly & Doubly Linked Lists

Store elements in non-contiguous heap nodes linked by references, trading $O(1)$ random array indexing for $O(1)$ pointer-rewiring insertions without element shifting.

1. Intuition & Interactive Pointer Manipulator

Maintaining both head and tail references makes addFirst and addLast run in O(1) time, while search and arbitrary remove must walk nodes sequentially in O(n) time.

2. Complexity Cheat Sheet

Operation Singly Linked (head only) Singly Linked (head & tail) Doubly Linked (head & tail)
Access by Index O(n) O(n) O(n)
Search by Value O(n) O(n) O(n)
Insert at Head O(1) O(1) O(1)
Insert at Tail O(n) O(1) O(1)
Delete at Head O(1) O(1) O(1)
Delete at Tail O(n) O(n) (needs prev) O(1)
Auxiliary Space O(n) O(n) O(n) (extra prev ref)

Why Singly Linked Tail Deletion is O(n): Even with a tail pointer, removing the final node requires finding the second-to-last node (prev) so prev.next can be set to null.

3. Step-by-Step Pseudocode & Operation Log

Delete First Matching Value

Pre: list has head & tail pointers; target is key to remove.

Post: Unlinks first matching node, updates head/tail, returns true or false.

DeleteValue(list, target)
    if list.head == null then
        return false
    if list.head.data == target then
        list.head ← list.head.next
        if list.head == null then
            list.tail ← null
        list.size ← list.size - 1
        return true

    prev ← list.head
    curr ← list.head.next
    while curr != null do
        if curr.data == target then
            prev.next ← curr.next
            if curr == list.tail then
                list.tail ← prev
            list.size ← list.size - 1
            return true
        prev ← curr
        curr ← curr.next
    return false

4. Java Implementation: Custom SinglyLinkedList vs. Built-in API

import java.util.Objects;

public class SinglyLinkedList {
    private static class Node {
        E data;
        Node next;

        Node(E data) {
            this.data = data;
        }
    }

    private Node head;
    private Node tail;
    private int size = 0;

    public void addFirst(E value) {
        Node newNode = new Node<>(value);
        newNode.next = head;
        head = newNode;
        if (tail == null) {
            tail = head;
        }
        size++;
    }

    public void addLast(E value) {
        Node newNode = new Node<>(value);
        if (tail == null) {
            head = tail = newNode;
        } else {
            tail.next = newNode;
            tail = newNode;
        }
        size++;
    }

    public boolean contains(E target) {
        Node curr = head;
        while (curr != null) {
            if (Objects.equals(curr.data, target)) {
                return true;
            }
            curr = curr.next;
        }
        return false;
    }

    public boolean remove(E target) {
        if (head == null) return false;

        if (Objects.equals(head.data, target)) {
            head = head.next;
            if (head == null) {
                tail = null;
            }
            size--;
            return true;
        }

        Node prev = head;
        Node curr = head.next;
        while (curr != null) {
            if (Objects.equals(curr.data, target)) {
                prev.next = curr.next;
                if (curr == tail) {
                    tail = prev;
                }
                size--;
                return true;
            }
            prev = curr;
            curr = curr.next;
        }
        return false;
    }

    public int size() {
        return size;
    }
}
import java.util.LinkedList;

public class BuiltInLinkedListDemo {
    public static void main(String[] args) {
        // Java's LinkedList is a doubly-linked list maintaining first & last pointers
        LinkedList list = new LinkedList<>();

        list.addFirst(10); // O(1)
        list.addLast(20);  // O(1)
        list.addLast(30);  // O(1)

        boolean found = list.contains(20); // O(n) -> true

        // IMPORTANT: Passing primitive int calls remove(int index).
        // Wrap in Integer.valueOf() to call remove(Object o) by value!
        list.remove(Integer.valueOf(20)); // O(n) -> removes value 20
        list.removeFirst();               // O(1) -> removes 10
        list.removeLast();                // O(1) -> removes 30 (O(1) because doubly linked)
    }
}
