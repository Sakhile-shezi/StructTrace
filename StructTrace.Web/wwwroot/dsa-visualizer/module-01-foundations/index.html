3. Step-by-Step Pseudocode & Live Trace

Iterative Binary Search

Pre: Sorted array A of size n; search key target.

Post: Returns index i where A[i] == target, or -1.

BinarySearch(A, n, target)
    low ← 0
    high ← n - 1
    while low ≤ high do
        mid ← low + floor((high - low) / 2)
        if A[mid] == target then
            return mid
        else if A[mid] < target then
            low ← mid + 1
        else
            high ← mid - 1
    return -1

4. Java Implementation: Built-in API vs. Custom Generic

public class GenericSearch {
    public static > int binarySearch(T[] array, T target) {
        int low = 0;
        int high = array.length - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2; // Prevents 32-bit int overflow
            int cmp = array[mid].compareTo(target);

            if (cmp == 0) {
                return mid;
            } else if (cmp < 0) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        return -1;
    }
}
import java.util.Arrays;

public class BuiltInSearchDemo {
    public static int search(Integer[] sortedData, Integer target) {
        // Returns index >= 0 if found, or -(insertionPoint) - 1 if absent
        int index = Arrays.binarySearch(sortedData, target);
        return index >= 0 ? index : -1;
    }
}
