Module 02 • Linear Pointer Structures

Singly, Doubly & Circular Linked Lists

Store elements in non-contiguous heap nodes linked by object references. Compare unidirectional next chains against bidirectional prev/next links and ring-shaped tail.next → head circular lists.

2. Singly vs. Doubly vs. Circular Complexity Cheat Sheet

Operation Singly (head & tail) Doubly (head & tail) Circular Singly (tail ref) Pointer Overhead per Node
addFirst(e) O(1) O(1) O(1) (tail.next = node) Singly / Circular: 1 ref
addLast(e) O(1) O(1) O(1) (advance tail) Doubly: 2 refs (prev, next)
removeFirst() O(1) O(1) O(1) (tail.next = head.next) —
removeLast() O(n) (no prev link) O(1) (tail = tail.prev) O(n) (unless Circular Doubly) —
rotate() (Round-Robin) O(n) (re-link ends) O(1) (move head to tail) O(1) (tail = tail.next) Zero memory allocations!
contains(e) / get(i) O(n) O(n) (halved from closer end) O(n) (stop when back at head) —

Why Circular Lists Store Only tail: In a Circular Singly Linked List, keeping a single reference to tail gives instant O(1) access to both ends because head is always tail.next! Calling rotate() is a single assignment: tail = tail.next.

3. Step-by-Step Pseudocode & Pointer Trace

Doubly removeLast vs. Circular rotate

Pre: List is non-empty (size > 0).

Post: Updates boundary pointers in O(1) time without scanning the list.

// Doubly Linked List: O(1) removeLast
DoublyRemoveLast(list)
    if list.size == 0 then error "Underflow"
    removed ← list.tail.data
    if list.size == 1 then
        list.head ← null
        list.tail ← null
    else
        list.tail ← list.tail.prev
        list.tail.next ← null
    list.size ← list.size - 1
    return removed

// Circular Singly List: O(1) rotate
CircularRotate(list)
    if list.tail != null then
        list.tail ← list.tail.next

4. Java Implementation: Singly, Doubly, Circular & Built-in LinkedList

import java.util.NoSuchElementException;

public class SinglyLinkedList {
    private static class Node {
        E data;
        Node next;
        Node(E data, Node next) {
            this.data = data;
            this.next = next;
        }
    }

    private Node head;
    private Node tail;
    private int size = 0;

    public void addFirst(E element) {
        head = new Node<>(element, head);
        if (size == 0) tail = head;
        size++;
    }

    public void addLast(E element) {
        Node newest = new Node<>(element, null);
        if (size == 0) head = newest;
        else tail.next = newest;
        tail = newest;
        size++;
    }

    public E removeFirst() {
        if (size == 0) throw new NoSuchElementException("List is empty");
        E answer = head.data;
        head = head.next;
        size--;
        if (size == 0) tail = null;
        return answer;
    }

    // O(n): Must traverse to locate node immediately before tail
    public E removeLast() {
        if (size == 0) throw new NoSuchElementException("List is empty");
        if (size == 1) return removeFirst();

        Node prev = head;
        while (prev.next != tail) {
            prev = prev.next;
        }
        E answer = tail.data;
        prev.next = null;
        tail = prev;
        size--;
        return answer;
    }
}
import java.util.NoSuchElementException;

public class DoublyLinkedList {
    private static class Node {
        E data;
        Node prev;
        Node next;
        Node(E data, Node prev, Node next) {
            this.data = data;
            this.prev = prev;
            this.next = next;
        }
    }

    // Sentinel dummy nodes eliminate null boundary checks
    private final Node header;
    private final Node trailer;
    private int size = 0;

    public DoublyLinkedList() {
        header = new Node<>(null, null, null);
        trailer = new Node<>(null, header, null);
        header.next = trailer;
    }

    public void addFirst(E e) {
        addBetween(e, header, header.next);
    }

    public void addLast(E e) {
        addBetween(e, trailer.prev, trailer);
    }

    public E removeFirst() {
        if (size == 0) throw new NoSuchElementException("List is empty");
        return remove(header.next);
    }

    // O(1) removal via trailer.prev!
    public E removeLast() {
        if (size == 0) throw new NoSuchElementException("List is empty");
        return remove(trailer.prev);
    }

    private void addBetween(E e, Node predecessor, Node successor) {
        Node newest = new Node<>(e, predecessor, successor);
        predecessor.next = newest;
        successor.prev = newest;
        size++;
    }

    private E remove(Node node) {
        Node predecessor = node.prev;
        Node successor = node.next;
        predecessor.next = successor;
        successor.prev = predecessor;
        size--;
        return node.data;
    }
}
import java.util.NoSuchElementException;

public class CircularLinkedList {
    private static class Node {
        E data;
        Node next;
        Node(E data, Node next) {
            this.data = data;
            this.next = next;
        }
    }

    // Only tail is stored; head is always tail.next
    private Node tail = null;
    private int size = 0;

    public E first() {
        return (size == 0) ? null : tail.next.data;
    }

    public E last() {
        return (size == 0) ? null : tail.data;
    }

    // O(1) round-robin rotation
    public void rotate() {
        if (tail != null) {
            tail = tail.next;
        }
    }

    public void addFirst(E e) {
        if (size == 0) {
            tail = new Node<>(e, null);
            tail.next = tail; // Points to itself
        } else {
            Node newest = new Node<>(e, tail.next);
            tail.next = newest;
        }
        size++;
    }

    public void addLast(E e) {
        addFirst(e);
        tail = tail.next; // New node becomes tail
    }

    public E removeFirst() {
        if (size == 0) throw new NoSuchElementException("Circular list is empty");
        Node head = tail.next;
        if (head == tail) {
            tail = null; // Removed sole node
        } else {
            tail.next = head.next;
        }
        size--;
        return head.data;
    }
}
import java.util.LinkedList;

public class BuiltInLinkedListDemo {
    public static void main(String[] args) {
        // Java's LinkedList is a Doubly Linked List implementing List & Deque
        LinkedList list = new LinkedList<>();
        list.addLast(10);
        list.addLast(20);
        list.addFirst(5);    // [5, 10, 20] -> O(1)

        int first = list.removeFirst(); // 5  -> O(1)
        int last = list.removeLast();   // 20 -> O(1) (uses tail.prev)
        boolean found = list.contains(10); // true -> O(n)
    }
}
