Binary Search Trees (BST) & Traversals

Maintain the ordering invariant left.key < node.key < right.key at every node so each comparison eliminates an entire subtree—achieving O(log n) search, insertion, and deletion when balanced, or degrading to O(n) when skewed.

2. Complexity Cheat Sheet

Operation Average Case (Balanced) Worst Case (Skewed) Auxiliary Space
contains(key) O(log n) O(n) O(1) iterative / O(h) recursive
insert(key) O(log n) O(n) O(h) call stack
delete(key) O(log n) O(n) O(h) call stack
findMin() / findMax() O(log n) O(n) O(1) iterative
Inorder / Preorder / Postorder O(n) O(n) O(h) recursion stack
Breadth-First (Level Order) O(n) O(n) O(w) queue (w = max width)

Why Skewness Happens: A standard BST performs no structural rebalancing. Inserting keys in sorted order (10, 20, 30, 40, 50) creates a right-leaning linked chain of height h = n - 1, motivating self-balancing AVL and Red-Black trees in Module 6.

3. Step-by-Step Pseudocode & Execution Log

BST Deletion (All 4 Cases)

Pre: root is the root of a valid BST (or null); key is target value.

Post: Returns updated subtree root with key removed and BST invariant preserved.

Delete(root, key)
    if root == null then
        return null

    if key < root.key then
        root.left ← Delete(root.left, key)
    else if key > root.key then
        root.right ← Delete(root.right, key)
    else
        // Case 1 & 2: 0 or 1 child
        if root.left == null then
            return root.right
        else if root.right == null then
            return root.left

        // Case 3: 2 children → Inorder Successor
        successor ← FindMin(root.right)
        root.key ← successor.key
        root.right ← Delete(root.right, successor.key)

    return root

FindMin(node)
    while node.left != null do
        node ← node.left
    return node

4. Java Implementation: Custom BinarySearchTree vs. Built-in TreeSet

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.List;
import java.util.Queue;

public class BinarySearchTree> {
    private static class Node {
        E key;
        Node left;
        Node right;

        Node(E key) {
            this.key = key;
        }
    }

    private Node root;

    public void insert(E key) {
        if (key == null) throw new IllegalArgumentException("Null keys are not permitted");
        root = insertRec(root, key);
    }

    private Node insertRec(Node node, E key) {
        if (node == null) return new Node<>(key);
        int cmp = key.compareTo(node.key);
        if (cmp < 0) {
            node.left = insertRec(node.left, key);
        } else if (cmp > 0) {
            node.right = insertRec(node.right, key);
        }
        return node; // Duplicate keys ignored
    }

    public boolean contains(E key) {
        if (key == null) return false;
        Node curr = root;
        while (curr != null) {
            int cmp = key.compareTo(curr.key);
            if (cmp == 0) return true;
            curr = (cmp < 0) ? curr.left : curr.right;
        }
        return false;
    }

    public void delete(E key) {
        if (key == null) return;
        root = deleteRec(root, key);
    }

    private Node deleteRec(Node node, E key) {
        if (node == null) return null;

        int cmp = key.compareTo(node.key);
        if (cmp < 0) {
            node.left = deleteRec(node.left, key);
        } else if (cmp > 0) {
            node.right = deleteRec(node.right, key);
        } else {
            // Case 1 & 2: 0 or 1 child
            if (node.left == null) return node.right;
            if (node.right == null) return node.left;

            // Case 3: 2 children → replace with Inorder Successor
            Node successor = findMin(node.right);
            node.key = successor.key;
            node.right = deleteRec(node.right, successor.key);
        }
        return node;
    }

    private Node findMin(Node node) {
        while (node.left != null) {
            node = node.left;
        }
        return node;
    }

    // Yields keys in sorted ascending order
    public List inOrderTraversal() {
        List result = new ArrayList<>();
        inOrderRec(root, result);
        return result;
    }

    private void inOrderRec(Node node, List result) {
        if (node == null) return;
        inOrderRec(node.left, result);
        result.add(node.key);
        inOrderRec(node.right, result);
    }

    // Level-by-level traversal using a FIFO queue
    public List breadthFirstTraversal() {
        List result = new ArrayList<>();
        if (root == null) return result;

        Queue> queue = new ArrayDeque<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            Node curr = queue.poll();
            result.add(curr.key);
            if (curr.left != null) queue.offer(curr.left);
            if (curr.right != null) queue.offer(curr.right);
        }
        return result;
    }
}
import java.util.TreeSet;

public class BuiltInTreeSetDemo {
    public static void main(String[] args) {
        // TreeSet wraps a self-balancing Red-Black Tree (guaranteed O(log n))
        TreeSet bst = new TreeSet<>();
        bst.add(50);
        bst.add(30);
        bst.add(70);
        bst.add(20);
        bst.add(40);
        bst.add(60);
        bst.add(80);

        boolean exists = bst.contains(40); // O(log n) -> true
        int min = bst.first();             // O(log n) -> 20
        int max = bst.last();              // O(log n) -> 80

        // Ordered navigation queries
        Integer successorOf50 = bst.higher(50);   // 60
        Integer predecessorOf50 = bst.lower(50);  // 40

        bst.remove(50); // O(log n) -> rebalances automatically
        System.out.println(bst); // [20, 30, 40, 60, 70, 80] (Inorder)
    }
}
