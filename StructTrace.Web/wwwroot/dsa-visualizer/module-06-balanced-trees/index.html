Module 06 • Self-Balancing Search Trees

AVL Trees, Rotations & Red-Black Invariants

Eliminate worst-case O(n) BST skewness by tracking each node's Balance Factor (BF = height(left) − height(right)) and applying O(1) pointer rotations whenever |BF| > 1.

2. Balanced Tree Comparison & Complexity Cheat Sheet

Tree Variant Search (Worst) Insert (Worst) Delete (Worst) Balancing Rule Rotations per Update
Unbalanced BST O(n) O(n) O(n) None (degenerates on sorted input) 0
AVL Tree O(log n) O(log n) O(log n) |height(L) − height(R)| ≤ 1 Insert: ≤ 2 • Delete: O(log n)
Red-Black Tree O(log n) O(log n) O(log n) Equal black-height on all root-to-null paths Insert: ≤ 2 • Delete: ≤ 3 (O(1))
2-4 Tree / B-Tree O(log n) O(log n) O(log n) Multi-key nodes; all leaves at exact same depth Node splits / merges (disk & DB friendly)

AVL vs. Red-Black in Practice: AVL trees maintain stricter height balance ($h < 1.44 \log_2 n$), making read-heavy lookups slightly faster. Red-Black trees allow slightly looser height ($h \le 2 \log_2 n$) to cap rotations at $O(1)$ during deletions—which is why java.util.TreeMap and java.util.TreeSet use Red-Black trees.

3. Step-by-Step Pseudocode & Rotation Log

AVL Rotations & Rebalance

Pre: y is an unbalanced node whose left child x is non-null.

Post: Performs a right rotation around y, updates heights bottom-up, and returns new root x.

RotateRight(y)
    x ← y.left
    T2 ← x.right

    // Perform rotation
    x.right ← y
    y.left ← T2

    // Update heights (child y first, then new root x)
    y.height ← 1 + max(Height(y.left), Height(y.right))
    x.height ← 1 + max(Height(x.left), Height(x.right))
    return x

Rebalance(node)
    UpdateHeight(node)
    bf ← Height(node.left) - Height(node.right)

    // Left-Heavy (bf == +2)
    if bf > 1 then
        if BalanceFactor(node.left) < 0 then
            node.left ← RotateLeft(node.left) // LR Case
        return RotateRight(node)              // LL Case

    // Right-Heavy (bf == -2)
    if bf < -1 then
        if BalanceFactor(node.right) > 0 then
            node.right ← RotateRight(node.right) // RL Case
        return RotateLeft(node)                // RR Case

    return node

4. Java Implementation: Custom AVLTree vs. Built-in TreeMap

public class AVLTree> {
    private static class Node {
        E key;
        Node left, right;
        int height = 0; // Leaf node has height 0; null has height -1

        Node(E key) {
            this.key = key;
        }
    }

    private Node root;

    private int height(Node n) {
        return (n == null) ? -1 : n.height;
    }

    private void updateHeight(Node n) {
        n.height = 1 + Math.max(height(n.left), height(n.right));
    }

    private int getBalance(Node n) {
        return (n == null) ? 0 : height(n.left) - height(n.right);
    }

    private Node rotateRight(Node y) {
        Node x = y.left;
        Node t2 = x.right;

        x.right = y;
        y.left = t2;

        updateHeight(y);
        updateHeight(x);
        return x;
    }

    private Node rotateLeft(Node x) {
        Node y = x.right;
        Node t2 = y.left;

        y.left = x;
        x.right = t2;

        updateHeight(x);
        updateHeight(y);
        return y;
    }

    private Node rebalance(Node node) {
        updateHeight(node);
        int bf = getBalance(node);

        // Left-Heavy
        if (bf > 1) {
            if (getBalance(node.left) < 0) {
                node.left = rotateLeft(node.left); // Left-Right (LR)
            }
            return rotateRight(node);              // Left-Left (LL)
        }

        // Right-Heavy
        if (bf < -1) {
            if (getBalance(node.right) > 0) {
                node.right = rotateRight(node.right); // Right-Left (RL)
            }
            return rotateLeft(node);                // Right-Right (RR)
        }
        return node;
    }

    public void insert(E key) {
        if (key == null) throw new IllegalArgumentException("Null keys not allowed");
        root = insertRec(root, key);
    }

    private Node insertRec(Node node, E key) {
        if (node == null) return new Node<>(key);
        int cmp = key.compareTo(node.key);
        if (cmp < 0) node.left = insertRec(node.left, key);
        else if (cmp > 0) node.right = insertRec(node.right, key);
        else return node; // Duplicate ignored

        return rebalance(node);
    }

    public void delete(E key) {
        if (key == null) return;
        root = deleteRec(root, key);
    }

    private Node deleteRec(Node node, E key) {
        if (node == null) return null;
        int cmp = key.compareTo(node.key);
        if (cmp < 0) {
            node.left = deleteRec(node.left, key);
        } else if (cmp > 0) {
            node.right = deleteRec(node.right, key);
        } else {
            if (node.left == null) return node.right;
            if (node.right == null) return node.left;

            Node succ = node.right;
            while (succ.left != null) succ = succ.left;
            node.key = succ.key;
            node.right = deleteRec(node.right, succ.key);
        }
        return rebalance(node);
    }
}
import java.util.NavigableMap;
import java.util.TreeMap;

public class BuiltInBalancedTreeDemo {
    public static void main(String[] args) {
        // TreeMap is backed by a self-balancing Red-Black Tree (O(log n) worst-case)
        NavigableMap scores = new TreeMap<>();
        scores.put(30, "Bronze");
        scores.put(10, "Starter");
        scores.put(20, "Silver");
        scores.put(40, "Gold");

        // Guaranteed O(log n) boundary & range queries
        System.out.println("Lowest key:  " + scores.firstKey());       // 10
        System.out.println("Ceiling(25): " + scores.ceilingKey(25));   // 30
        System.out.println("Floor(25):   " + scores.floorKey(25));     // 20
        System.out.println("SubMap 15..35: " + scores.subMap(15, 35)); // {20=Silver, 30=Bronze}
    }
}
