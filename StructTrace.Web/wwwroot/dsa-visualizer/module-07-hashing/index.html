Sets, Maps, Hash Functions & Collision Resolution

Map arbitrary keys directly to array bucket indices via index = (key.hashCode() & 0x7fffffff) % M, achieving amortized O(1) lookup, insertion, and deletion while resolving collisions through Separate Chaining or Open Addressing.

Why Mask With 0x7fffffff? Java's hashCode() returns a signed 32-bit int that can be negative (e.g., "polygenelubricants".hashCode() is Integer.MIN_VALUE). Applying (hash & 0x7fffffff) % M clears the sign bit safely before taking the modulo so array indices are never negative.

3. Step-by-Step Pseudocode & Collision Log

Separate Chaining Put & Rehash

Pre: table has bucket array B[0..M-1]; key is non-null.

Post: Inserts or updates (key, value) and doubles/rehashes if size / M ≥ 0.75.

Put(table, key, value)
    idx ← (HashCode(key) & 0x7fffffff) mod table.M
    curr ← table.B[idx]

    while curr != null do
        if curr.key == key then
            curr.value ← value
            return
        curr ← curr.next

    // Prepend new node to bucket chain
    node ← new Entry(key, value, next = table.B[idx])
    table.B[idx] ← node
    table.size ← table.size + 1

    if (table.size / table.M) ≥ 0.75 then
        Rehash(table, NextPrime(2 * table.M))

4. Java Implementation: Custom ChainedHashMap vs. Built-in HashMap & Set Ops

import java.util.Objects;

public class ChainedHashMap {
    private static class Entry {
        final K key;
        V value;
        Entry next;

        Entry(K key, V value, Entry next) {
            this.key = key;
            this.value = value;
            this.next = next;
        }
    }

    private static final double MAX_LOAD_FACTOR = 0.75;
    private Entry[] buckets;
    private int size = 0;

    @SuppressWarnings("unchecked")
    public ChainedHashMap(int initialCapacity) {
        this.buckets = (Entry[]) new Entry[initialCapacity];
    }

    private int indexFor(K key, int capacity) {
        int hash = (key == null) ? 0 : key.hashCode();
        return (hash & 0x7fffffff) % capacity;
    }

    public V get(K key) {
        int idx = indexFor(key, buckets.length);
        Entry curr = buckets[idx];
        while (curr != null) {
            if (Objects.equals(curr.key, key)) {
                return curr.value;
            }
            curr = curr.next;
        }
        return null;
    }

    public void put(K key, V value) {
        int idx = indexFor(key, buckets.length);
        Entry curr = buckets[idx];
        while (curr != null) {
            if (Objects.equals(curr.key, key)) {
                curr.value = value; // Update existing key
                return;
            }
            curr = curr.next;
        }

        buckets[idx] = new Entry<>(key, value, buckets[idx]);
        size++;

        if ((double) size / buckets.length >= MAX_LOAD_FACTOR) {
            rehash(buckets.length * 2 + 1);
        }
    }

    public V remove(K key) {
        int idx = indexFor(key, buckets.length);
        Entry prev = null;
        Entry curr = buckets[idx];

        while (curr != null) {
            if (Objects.equals(curr.key, key)) {
                if (prev == null) buckets[idx] = curr.next;
                else prev.next = curr.next;
                size--;
                return curr.value;
            }
            prev = curr;
            curr = curr.next;
        }
        return null;
    }

    @SuppressWarnings("unchecked")
    private void rehash(int newCapacity) {
        Entry[] oldBuckets = buckets;
        buckets = (Entry[]) new Entry[newCapacity];

        for (Entry head : oldBuckets) {
            Entry curr = head;
            while (curr != null) {
                Entry next = curr.next;
                int newIdx = indexFor(curr.key, newCapacity);
                curr.next = buckets[newIdx];
                buckets[newIdx] = curr;
                curr = next;
            }
        }
    }

    public int size() {
        return size;
    }
}
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class BuiltInHashDemo {
    public static void main(String[] args) {
        // 1. O(1) Frequency Map with HashMap
        Map grades = new HashMap<>();
        grades.put("Alice", 92);
        grades.put("Bob", 85);
        int bobScore = grades.getOrDefault("Bob", 0); // 85

        // 2. Mathematical Set Operations in O(m + n) expected time
        Set cs201 = new HashSet<>(List.of("Alice", "Bob", "Cleo"));
        Set cs315 = new HashSet<>(List.of("Bob", "Cleo", "Dan"));

        // Union (cs201 ∪ cs315)
        Set union = new HashSet<>(cs201);
        union.addAll(cs315); // [Alice, Bob, Cleo, Dan]

        // Intersection (cs201 ∩ cs315)
        Set intersection = new HashSet<>(cs201);
        intersection.retainAll(cs315); // [Bob, Cleo]

        // Difference (cs201 \ cs315)
        Set difference = new HashSet<>(cs201);
        difference.removeAll(cs315); // [Alice]
    }
}
