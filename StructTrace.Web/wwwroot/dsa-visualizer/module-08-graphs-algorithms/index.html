2. Graph Representations & Traversal Cheat Sheet

Representation / Algorithm Space Check Edge (u, v) Iterate Neighbors of u Primary Use Case
Adjacency List O(V + E) O(deg(u)) O(deg(u)) Sparse graphs (road networks, web links, social graphs)
Adjacency Matrix O(V²) O(1) O(V) Dense graphs or frequent constant-time edge lookups
Breadth-First Search (BFS) O(V) queue Total runtime: O(V + E) (list) / O(V²) (matrix) Shortest path by edge count in unweighted graphs
Depth-First Search (DFS) O(V) stack Total runtime: O(V + E) (list) / O(V²) (matrix) Cycle detection, topological sorting, connected components

Why BFS Finds Unweighted Shortest Paths: Because a FIFO queue processes all vertices at distance $k$ before any vertex at distance $k + 1$, the first time BFS reaches vertex $v$ is guaranteed to follow a path with the minimum possible number of edges.

3. Step-by-Step Pseudocode & Execution Log

BFS (Queue) & Iterative DFS (Stack)

Pre: Start vertex $s \in V$ exists in graph $G = (V, E)$.

Post: Visits every vertex reachable from $s$ exactly once in O(V + E) time.

BFS(G, s)
    visited ← { s }
    Q ← empty Queue
    Q.enqueue(s)
    while not Q.isEmpty() do
        u ← Q.dequeue()
        for each v in G.neighbors(u) do
            if v not in visited then
                visited.add(v)
                Q.enqueue(v)

DFS(G, s)
    visited ← empty Set
    S ← empty Stack
    S.push(s)
    while not S.isEmpty() do
        u ← S.pop()
        if u not in visited then
            visited.add(u)
            for each v in reverse(G.neighbors(u)) do
                if v not in visited then
                    S.push(v)

4. Java Implementation: Generic Adjacency List & Matrix

import java.util.*;

public class Graph {
    private final Map> adj = new LinkedHashMap<>();
    private final boolean directed;

    public Graph(boolean directed) {
        this.directed = directed;
    }

    public void addVertex(V v) {
        adj.putIfAbsent(v, new ArrayList<>());
    }

    public void addEdge(V u, V v) {
        addVertex(u);
        addVertex(v);
        adj.get(u).add(v);
        if (!directed) {
            adj.get(v).add(u);
        }
    }

    // Level-order traversal using FIFO ArrayDeque
    public List bfs(V start) {
        List visitedOrder = new ArrayList<>();
        if (!adj.containsKey(start)) return visitedOrder;

        Set visited = new HashSet<>(List.of(start));
        Queue queue = new ArrayDeque<>();
        queue.offer(start);

        while (!queue.isEmpty()) {
            V curr = queue.poll();
            visitedOrder.add(curr);
            for (V neighbor : adj.get(curr)) {
                if (visited.add(neighbor)) {
                    queue.offer(neighbor);
                }
            }
        }
        return visitedOrder;
    }

    // Depth-first traversal using LIFO ArrayDeque
    public List dfs(V start) {
        List visitedOrder = new ArrayList<>();
        if (!adj.containsKey(start)) return visitedOrder;

        Set visited = new HashSet<>();
        Deque stack = new ArrayDeque<>();
        stack.push(start);

        while (!stack.isEmpty()) {
            V curr = stack.pop();
            if (visited.add(curr)) {
                visitedOrder.add(curr);
                List neighbors = adj.get(curr);
                for (int i = neighbors.size() - 1; i >= 0; i--) {
                    V next = neighbors.get(i);
                    if (!visited.contains(next)) {
                        stack.push(next);
                    }
                }
            }
        }
        return visitedOrder;
    }
}
import java.util.*;

public class MatrixGraph {
    private final int numVertices;
    private final boolean[][] matrix;
    private final boolean directed;

    public MatrixGraph(int numVertices, boolean directed) {
        this.numVertices = numVertices;
        this.directed = directed;
        this.matrix = new boolean[numVertices][numVertices];
    }

    public void addEdge(int u, int v) {
        matrix[u][v] = true;
        if (!directed) {
            matrix[v][u] = true;
        }
    }

    // O(1) constant-time adjacency query
    public boolean hasEdge(int u, int v) {
        return matrix[u][v];
    }

    // O(V) neighbor scan per vertex -> O(V^2) total BFS runtime
    public List bfs(int start) {
        List order = new ArrayList<>();
        boolean[] visited = new boolean[numVertices];
        Queue queue = new ArrayDeque<>();

        visited[start] = true;
        queue.offer(start);

        while (!queue.isEmpty()) {
            int u = queue.poll();
            order.add(u);
            for (int v = 0; v < numVertices; v++) {
                if (matrix[u][v] && !visited[v]) {
                    visited[v] = true;
                    queue.offer(v);
                }
            }
        }
        return order;
    }
}
