3. Java Implementation: Dijkstra & MergeSort

import java.util.*;

public class DijkstraSolver {
    public record Edge(V target, int weight) {}
    private record State(V vertex, int dist) implements Comparable> {
        public int compareTo(State other) {
            return Integer.compare(this.dist, other.dist);
        }
    }

    public Map shortestPaths(Map>> adj, V source) {
        Map dist = new HashMap<>();
        for (V v : adj.keySet()) dist.put(v, Integer.MAX_VALUE);
        dist.put(source, 0);

        PriorityQueue> pq = new PriorityQueue<>();
        pq.offer(new State<>(source, 0));

        while (!pq.isEmpty()) {
            State curr = pq.poll();
            if (curr.dist() > dist.get(curr.vertex())) continue;

            for (Edge e : adj.getOrDefault(curr.vertex(), List.of())) {
                int candidate = curr.dist() + e.weight();
                if (candidate < dist.get(e.target())) {
                    dist.put(e.target(), candidate);
                    pq.offer(new State<>(e.target(), candidate));
                }
            }
        }
        return dist;
    }
}
import java.util.Arrays;

public class MergeSort {
    public static void sort(int[] arr) {
        if (arr.length < 2) return;
        int mid = arr.length / 2;
        int[] left = Arrays.copyOfRange(arr, 0, mid);
        int[] right = Arrays.copyOfRange(arr, mid, arr.length);

        sort(left);
        sort(right);
        merge(arr, left, right);
    }

    private static void merge(int[] dest, int[] left, int[] right) {
        int i = 0, j = 0, k = 0;
        while (i < left.length && j < right.length) {
            dest[k++] = (left[i] <= right[j]) ? left[i++] : right[j++];
        }
        while (i < left.length) dest[k++] = left[i++];
        while (j < right.length) dest[k++] = right[j++];
    }
}
